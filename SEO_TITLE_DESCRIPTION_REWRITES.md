## 2. Dynamic IPO Detail Route Template & Examples (`/ipo/[slug]/[symbol]`)

### Metadata Generation Formula
```typescript
// Proposed High-CTR Formula in src/lib/seo.ts:
export function getIpoMetadata(ipo: Ipo): Metadata {
  const pct = gmpPercent(ipo);
  const subTotal = ipo.subscription?.total || 0;
  const canonicalUrl = `${SITE_URL}/ipo/${ipo.slug}/${ipo.symbol}`;

  // Title: max ~44 chars base so base + " | ipo-premium" <= 58 chars
  // Format: "{Name} IPO GMP ₹{gmp} ({+pct}%)"
  const shortName = ipo.name.length > 20 ? ipo.name.slice(0, 19) + "…" : ipo.name;
  const title = `${shortName} IPO GMP ₹${ipo.gmp} (${formatPercent(pct)})`;

  // Description: exactly 145-155 characters
  // Format: "{Name} IPO live GMP today is ₹{gmp} ({+pct}% gain). Price band: {band}, lot: {lot}. Subscribed {sub}x. Check allotment date & review."
  const description = `${ipo.name} IPO live GMP today is ₹${ipo.gmp} (${formatPercent(pct)} expected gain). Price band: ${ipo.priceBand}, lot: ${ipo.lotSize}. Subscribed ${formatSubscription(subTotal)}. Check allotment dates & review.`;

  return {
    title,
    description: description.slice(0, 155),
    alternates: { canonical: canonicalUrl },
    openGraph: { title, description, url: canonicalUrl, type: "article" },
    twitter: { card: "summary_large_image", title, description },
  };
}
```
